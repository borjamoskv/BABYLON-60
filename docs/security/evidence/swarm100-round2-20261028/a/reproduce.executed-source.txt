def audit(event, args):
    if event in ('subprocess.Popen', 'os.system', 'os.posix_spawn') or event.startswith('socket.'):
        blocked.append(event)
        raise RuntimeError('Forbidden external boundary: ' + event)
    if event == 'open':
        path, mode, flags = args
        if isinstance(path, (str, bytes, os.PathLike)) and (flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND)):
            p = pathlib.Path(os.fsdecode(path)).absolute()
            if not p.is_relative_to(BASE):
                raise RuntimeError('Write outside audit fixture refused')
            write_opens.append(str(p.relative_to(BASE)))
    if event in ('os.mkdir', 'os.rename', 'os.rmdir', 'os.remove'):
        p = args[0]
        # CPython shutil uses dir_fd for children of the reviewed absolute fixture root.
        if isinstance(p, (str, bytes, os.PathLike)) and os.path.isabs(p):
            if not pathlib.Path(os.fsdecode(p)).absolute().is_relative_to(BASE):
                raise RuntimeError('Filesystem mutation outside fixture refused')
sys.addaudithook(audit)

def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module
